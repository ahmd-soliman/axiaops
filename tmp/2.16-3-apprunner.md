# 2.16-3 — App Runner (api + ingestion)

> Deploy both Go services to App Runner. Depends on ECR (2.16-1), RDS (2.16-2), and Secrets Manager (2.16-5).
> Do Secrets Manager first, then reference secret ARNs here.

---

## What to build

- `axiaops-api` App Runner service — port 8080
- `axiaops-ingestion` App Runner service — port 8081
- `AxiaOpsAppRunnerRole` IAM role — ECR pull + Secrets Manager read + AWS scan APIs
- App Runner instance role — passed to the running container

---

## terraform/iam.tf

```hcl
data "aws_iam_policy_document" "apprunner_assume" {
  statement {
    actions = ["sts:AssumeRole"]
    principals {
      type        = "Service"
      identifiers = ["tasks.apprunner.amazonaws.com"]
    }
  }
}

resource "aws_iam_role" "apprunner" {
  name               = "AxiaOpsAppRunnerRole"
  assume_role_policy = data.aws_iam_policy_document.apprunner_assume.json
}

resource "aws_iam_role_policy" "apprunner" {
  role = aws_iam_role.apprunner.id
  policy = jsonencode({
    Version = "2012-10-17"
    Statement = [
      {
        Effect = "Allow"
        Action = [
          "secretsmanager:GetSecretValue"
        ]
        Resource = "arn:aws:secretsmanager:${var.aws_region}:${var.aws_account_id}:secret:axiaops/prod/*"
      },
      {
        Effect = "Allow"
        Action = [
          "ce:GetCostAndUsage",
          "cloudwatch:GetMetricStatistics",
          "cloudwatch:ListMetrics",
          "ec2:DescribeInstances",
          "ec2:DescribeNatGateways",
          "ec2:DescribeAddresses",
          "rds:DescribeDBInstances",
          "lambda:ListFunctions",
          "elasticloadbalancing:DescribeLoadBalancers"
        ]
        Resource = "*"
      }
    ]
  })
}

# ECR access role — used by App Runner to pull images
data "aws_iam_policy_document" "apprunner_ecr_assume" {
  statement {
    actions = ["sts:AssumeRole"]
    principals {
      type        = "Service"
      identifiers = ["build.apprunner.amazonaws.com"]
    }
  }
}

resource "aws_iam_role" "apprunner_ecr" {
  name               = "AxiaOpsAppRunnerECRRole"
  assume_role_policy = data.aws_iam_policy_document.apprunner_ecr_assume.json
}

resource "aws_iam_role_policy_attachment" "apprunner_ecr" {
  role       = aws_iam_role.apprunner_ecr.name
  policy_arn = "arn:aws:iam::aws:policy/service-role/AWSAppRunnerServicePolicyForECRAccess"
}
```

## terraform/apprunner.tf

```hcl
locals {
  ecr_registry = "${var.aws_account_id}.dkr.ecr.${var.aws_region}.amazonaws.com"

  # Shared env vars for both services — read from Secrets Manager at runtime
  common_env = [
    {
      name  = "DATABASE_URL"
      value = "SECRETS_MANAGER"  # replaced below with secret reference
    }
  ]
}

resource "aws_apprunner_service" "api" {
  service_name = "axiaops-api"

  source_configuration {
    authentication_configuration {
      access_role_arn = aws_iam_role.apprunner_ecr.arn
    }
    image_repository {
      image_identifier      = "${local.ecr_registry}/axiaops-api:latest"
      image_repository_type = "ECR"
      image_configuration {
        port = "8080"
        runtime_environment_secrets = {
          DATABASE_URL            = aws_secretsmanager_secret.database_url.arn
          MIGRATION_DATABASE_URL  = aws_secretsmanager_secret.migration_database_url.arn
          ENCRYPTION_KEY          = aws_secretsmanager_secret.encryption_key.arn
          REDIS_URL               = aws_secretsmanager_secret.redis_url.arn
          KINDE_ISSUER            = aws_secretsmanager_secret.kinde_issuer.arn
          KINDE_CLIENT_ID         = aws_secretsmanager_secret.kinde_client_id.arn
        }
        runtime_environment_variables = {
          API_ADDR     = ":8080"
          INGESTION_URL = "https://${aws_apprunner_service.ingestion.service_url}"
          AWS_REGION   = var.aws_region
        }
      }
    }
    auto_deployments_enabled = false  # CI triggers deploy via update-service
  }

  instance_configuration {
    cpu    = "0.25 vCPU"
    memory = "0.5 GB"
    instance_role_arn = aws_iam_role.apprunner.arn
  }

  health_check_configuration {
    protocol            = "HTTP"
    path                = "/health"
    interval            = 10
    timeout             = 5
    healthy_threshold   = 1
    unhealthy_threshold = 3
  }
}

resource "aws_apprunner_service" "ingestion" {
  service_name = "axiaops-ingestion"

  source_configuration {
    authentication_configuration {
      access_role_arn = aws_iam_role.apprunner_ecr.arn
    }
    image_repository {
      image_identifier      = "${local.ecr_registry}/axiaops-ingestion:latest"
      image_repository_type = "ECR"
      image_configuration {
        port = "8081"
        runtime_environment_secrets = {
          DATABASE_URL           = aws_secretsmanager_secret.database_url.arn
          MIGRATION_DATABASE_URL = aws_secretsmanager_secret.migration_database_url.arn
          ENCRYPTION_KEY         = aws_secretsmanager_secret.encryption_key.arn
          REDIS_URL              = aws_secretsmanager_secret.redis_url.arn
          KINDE_ISSUER           = aws_secretsmanager_secret.kinde_issuer.arn
        }
        runtime_environment_variables = {
          AWS_REGION   = var.aws_region
          SCAN_INTERVAL = "60m"
        }
      }
    }
    auto_deployments_enabled = false
  }

  instance_configuration {
    cpu    = "0.5 vCPU"
    memory = "1 GB"
    instance_role_arn = aws_iam_role.apprunner.arn
  }

  health_check_configuration {
    protocol            = "HTTP"
    path                = "/health"
    interval            = 10
    timeout             = 5
    healthy_threshold   = 1
    unhealthy_threshold = 3
  }
}
```

## GitLab CI variables to add after apply

| Variable | Value | Source |
|----------|-------|--------|
| `APPRUNNER_API_ARN` | from `terraform output api_service_arn` | yes |
| `APPRUNNER_INGESTION_ARN` | from `terraform output ingestion_service_arn` | yes |

The existing `deploy:production` job in `.gitlab-ci.yml` already calls `aws apprunner update-service` — it just needs the correct ARNs. Update the ARNs in the script to use the variables:

```yaml
# In .gitlab-ci.yml deploy:production script, replace hardcoded ARNs:
- aws apprunner update-service --region ${AWS_DEFAULT_REGION} \
    --service-arn ${APPRUNNER_API_ARN} \
    --source-configuration ImageRepository={...}
- aws apprunner update-service --region ${AWS_DEFAULT_REGION} \
    --service-arn ${APPRUNNER_INGESTION_ARN} \
    --source-configuration ImageRepository={...}
```

## First deploy sequence

```bash
# 1. Apply terraform (after secrets are created — see 2.16-5)
terraform apply -var-file=terraform.tfvars

# 2. Push initial images manually to unblock App Runner
aws ecr get-login-password --region eu-central-1 | \
  docker login --username AWS --password-stdin ACCOUNT.dkr.ecr.eu-central-1.amazonaws.com

docker build -f services/api/Dockerfile -t axiaops-api:latest .
docker tag axiaops-api:latest ACCOUNT.dkr.ecr.eu-central-1.amazonaws.com/axiaops-api:latest
docker push ACCOUNT.dkr.ecr.eu-central-1.amazonaws.com/axiaops-api:latest

docker build -f services/ingestion/Dockerfile -t axiaops-ingestion:latest .
docker tag axiaops-ingestion:latest ACCOUNT.dkr.ecr.eu-central-1.amazonaws.com/axiaops-ingestion:latest
docker push ACCOUNT.dkr.ecr.eu-central-1.amazonaws.com/axiaops-ingestion:latest

# 3. Verify health
curl https://API_URL/health
curl https://INGESTION_URL/health
```
