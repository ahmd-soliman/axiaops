DismissGhost(ctx context.Context, d model.DismissedGhost) error

// ClearExpiredSnoozes removes snooze records where snooze_until < NOW().

## PostgreSQL implementation (`postgres.go`)

### `DismissGhost`
```sql
INSERT INTO dismissed_ghosts (id, tenant_id, resource_id, reason, note, dismissed_by, dismissed_at, snooze_until)
VALUES ($1, $2, $3, $4, $5, $6, NOW(), $7)
ON CONFLICT (tenant_id, resource_id) DO UPDATE
  SET reason       = EXCLUDED.reason,
      note         = EXCLUDED.note,
      dismissed_by = EXCLUDED.dismissed_by,
      dismissed_at = NOW(),
      snooze_until = EXCLUDED.snooze_until
```

### `UndismissGhost`
```sql
DELETE FROM dismissed_ghosts
WHERE resource_id = $1
  AND tenant_id = current_setting('app.tenant_id', true)
```

### `ListDismissedResourceIDs`
```sql
SELECT resource_id FROM dismissed_ghosts
WHERE tenant_id = current_setting('app.tenant_id', true)
  AND (snooze_until IS NULL OR snooze_until > NOW())
```

### `ClearExpiredSnoozes`
```sql
DELETE FROM dismissed_ghosts
WHERE snooze_until IS NOT NULL AND snooze_until < NOW()
RETURNING id
```

### `dismissGhost`
```go
func (h *Handler) dismissGhost(w http.ResponseWriter, r *http.Request) {
    resourceID := r.PathValue("id")
    userID := middleware.UserID(r.Context())

    var req model.DismissRequest
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil || req.Reason == "" {
        http.Error(w, `{"error":"reason required"}`, http.StatusBadRequest)
        return
    }

    d := model.DismissedGhost{
        ID:          uuid.New().String(),
        ResourceID:  resourceID,
        Reason:      req.Reason,
        Note:        req.Note,
        DismissedBy: userID,
        SnoozeUntil: req.SnoozeUntil,
    }
    if err := h.store.DismissGhost(r.Context(), d); err != nil {
        http.Error(w, `{"error":"dismiss failed"}`, http.StatusInternalServerError)
        return
    }
    w.WriteHeader(http.StatusNoContent)
}
```

func (h *Handler) undismissGhost(w http.ResponseWriter, r *http.Request) {

}

### Update `listGhosts`

```go
func (h *Handler) listGhosts(w http.ResponseWriter, r *http.Request) {
    ghosts, err := h.store.LoadGhosts(r.Context())
    // ...
    includeDismissed := r.URL.Query().Get("include_dismissed") == "true"
    if !includeDismissed {
        dismissed, _ := h.store.ListDismissedResourceIDs(r.Context())
        filtered := ghosts[:0]
        for _, g := range ghosts {
            if !dismissed[g.ResourceID] {
                filtered = append(filtered, g)
            }
        }
        ghosts = filtered
    }
    writeJSON(w, ghosts)
}
```

## Background ticker — snooze expiry (`api/cmd/main.go`)

```go
// Runs every 15 minutes — clears expired snoozes
go func() {
    ticker := time.NewTicker(15 * time.Minute)
    defer ticker.Stop()
    for {
        select {
        case <-ctx.Done():
            return
        case <-ticker.C:
            n, err := store.ClearExpiredSnoozes(context.Background())
            if err != nil {
                slog.Error("dismiss: snooze cleanup failed", "error", err)
            } else if n > 0 {
                slog.Info("dismiss: cleared expired snoozes", "count", n)
            }
        }
    }
}()
```
