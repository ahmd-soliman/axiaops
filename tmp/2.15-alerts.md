# 2.15 — Weekly Email Digest + Slack Alerts

> Not a hard blocker for launch, but a core retention driver.
> Build after 2.16 (production deployment) so you have real users to send to.

---

## Overview

| Feature | Trigger | Recipient |
|---------|---------|-----------|
| Email digest | After scan, if ghost count changed vs last week | Tenant users (from `users` table) |
| Slack alert | After scan, if ghost count changed (new or resolved) | Per-account `slack_webhook_url` |

Both are fire-and-forget — failures are logged, never fatal to the scan.

```sql
    email_digest_enabled BOOLEAN     NOT NULL DEFAULT true,
    last_digest_sent_at  TIMESTAMPTZ
);
ALTER TABLE notification_settings ENABLE ROW LEVEL SECURITY;
CREATE POLICY notification_settings_isolation ON notification_settings
    USING (tenant_id = current_setting('app.tenant_id', true));
```

## New package: `services/shared/notify/`

### `notify/email.go` — Resend HTTP client

```go
package notify

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "net/http"
)

type DigestPayload struct {
    To              string
    GhostCount      int
    TotalMonthlyCost float64
    Currency        string
    WeekDelta       int     // positive = more ghosts, negative = fewer
    TopGhosts       []GhostSummary
}

type GhostSummary struct {
    Service     string
    ResourceID  string
    MonthlyCost float64
}

func SendDigest(ctx context.Context, apiKey string, p DigestPayload) error {
    body := map[string]any{
        "from":    "AxiaOps <alerts@axiaops.io>",
        "to":      []string{p.To},
        "subject": fmt.Sprintf("AxiaOps: %d ghost resources — $%.0f/mo potential savings", p.GhostCount, p.TotalMonthlyCost),
        "html":    buildDigestHTML(p),
    }
    b, _ := json.Marshal(body)
    req, _ := http.NewRequestWithContext(ctx, http.MethodPost, "https://api.resend.com/emails", bytes.NewReader(b))
    req.Header.Set("Authorization", "Bearer "+apiKey)
    req.Header.Set("Content-Type", "application/json")
    resp, err := http.DefaultClient.Do(req)
    if err != nil {
        return err
    }
    defer resp.Body.Close()
    if resp.StatusCode >= 300 {
        return fmt.Errorf("resend: status %d", resp.StatusCode)
    }
    return nil
}

func buildDigestHTML(p DigestPayload) string {
    // Minimal HTML — no template engine needed
    delta := ""
    if p.WeekDelta > 0 {
        delta = fmt.Sprintf("<p>⚠️ %d new ghost(s) since last week.</p>", p.WeekDelta)
    } else if p.WeekDelta < 0 {
        delta = fmt.Sprintf("<p>✅ %d ghost(s) resolved since last week.</p>", -p.WeekDelta)
    }
    rows := ""
    for _, g := range p.TopGhosts {
        rows += fmt.Sprintf("<tr><td>%s</td><td>%s</td><td>$%.2f/mo</td></tr>",
            g.Service, g.ResourceID, g.MonthlyCost)
    }
    return fmt.Sprintf(`<h2>AxiaOps Weekly Digest</h2>
<p>%d ghost resources detected — <strong>$%.2f %s/month</strong> potential savings.</p>
%s
<table border="1" cellpadding="4">
  <tr><th>Service</th><th>Resource</th><th>Cost</th></tr>
  %s
</table>
<p><a href="https://app.axiaops.io">View in dashboard →</a></p>`, p.GhostCount, p.TotalMonthlyCost, p.Currency, delta, rows)
}
```

### `notify/slack.go` — Slack webhook sender

```go
package notify

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "net/http"
)

func SendSlackAlert(ctx context.Context, webhookURL, message string) error {
    body, _ := json.Marshal(map[string]string{"text": message})
    req, _ := http.NewRequestWithContext(ctx, http.MethodPost, webhookURL, bytes.NewReader(body))
    req.Header.Set("Content-Type", "application/json")
    resp, err := http.DefaultClient.Do(req)
    if err != nil {
        return err
    }
    defer resp.Body.Close()
    if resp.StatusCode >= 300 {
        return fmt.Errorf("slack: status %d", resp.StatusCode)
    }
    return nil
}

func SlackMessage(accountLabel string, ghostCount int, totalCost float64, delta int) string {
    change := ""
    switch {
    case delta > 0:
        change = fmt.Sprintf(" (%d new)", delta)
    case delta < 0:
        change = fmt.Sprintf(" (%d resolved)", -delta)
    }
    return fmt.Sprintf("AxiaOps: *%d ghost resource(s)%s* in *%s* — *$%.0f/mo* potential savings.",
        ghostCount, change, accountLabel, totalCost)
}
```

## Wiring into `runIngestion` (`ingestion/cmd/main.go`)

After `store.SaveSnapshot(...)` succeeds, add:

```go
// Fire-and-forget notifications — never fatal to the scan
go func() {
    if err := sendNotifications(context.Background(), store, accountID, account.Label, ghosts, summary); err != nil {
        slog.Warn("notify: failed to send notifications", "account_id", accountID, "error", err)
    }
}()
```

```go
func sendNotifications(ctx context.Context, store storage.Store, accountID, accountLabel string, ghosts []model.GhostResource, summary analyzer.Summary) error {
    resendKey := os.Getenv("RESEND_API_KEY")
    
    // --- Slack ---
    account, err := store.GetAccount(ctx, accountID)
    if err == nil && account.SlackWebhookURL != "" {
        // Compute delta: current ghost count vs previous snapshot
        snaps, _ := store.ListSnapshots(ctx, accountID)
        delta := 0
        if len(snaps) >= 2 {
            delta = summary.TotalGhosts - snaps[len(snaps)-2].GhostCount
        }
        msg := notify.SlackMessage(accountLabel, summary.TotalGhosts, summary.PotentialMonthlySave, delta)
        if err := notify.SendSlackAlert(ctx, account.SlackWebhookURL, msg); err != nil {
            slog.Warn("notify: slack alert failed", "account_id", accountID, "error", err)
        }
    }

    // --- Email digest ---
    if resendKey == "" {
        return nil
    }
    settings, _ := store.GetNotificationSettings(ctx)
    if !settings.EmailDigestEnabled {
        return nil
    }
    // Frequency guard: skip if digest sent within last 6 days
    if settings.LastDigestSentAt != nil && time.Since(*settings.LastDigestSentAt) < 6*24*time.Hour {
        return nil
    }
    users, err := store.ListTenantUsers(ctx)
    if err != nil || len(users) == 0 {
        return err
    }
    // Top 5 ghosts by cost
    top := ghosts
    if len(top) > 5 {
        top = top[:5]
    }
    topSummary := make([]notify.GhostSummary, len(top))
    for i, g := range top {
        topSummary[i] = notify.GhostSummary{Service: g.Service, ResourceID: g.ResourceID, MonthlyCost: g.MonthlyCost}
    }
    snaps, _ := store.ListSnapshots(ctx, accountID)
    delta := 0
    if len(snaps) >= 2 {
        delta = summary.TotalGhosts - snaps[len(snaps)-2].GhostCount
    }
    for _, u := range users {
        if u.Email == "" {
            continue
        }
        _ = notify.SendDigest(ctx, resendKey, notify.DigestPayload{
            To:               u.Email,
            GhostCount:       summary.TotalGhosts,
            TotalMonthlyCost: summary.PotentialMonthlySave,
            Currency:         summary.Currency,
            WeekDelta:        delta,
            TopGhosts:        topSummary,
        })
    }
    // Update last sent timestamp
    now := time.Now().UTC()
    settings.LastDigestSentAt = &now
    _ = store.UpsertNotificationSettings(ctx, settings)
    return nil
}
```

## PostgreSQL implementation (`postgres.go`)

### `GetNotificationSettings`
```sql
SELECT tenant_id, email_digest_enabled, last_digest_sent_at
FROM notification_settings
WHERE tenant_id = current_setting('app.tenant_id', true)
```
Return defaults if no row found.

### `UpsertNotificationSettings`
```sql
INSERT INTO notification_settings (tenant_id, email_digest_enabled, last_digest_sent_at)
VALUES ($1, $2, $3)
ON CONFLICT (tenant_id) DO UPDATE
  SET email_digest_enabled = EXCLUDED.email_digest_enabled,
      last_digest_sent_at  = EXCLUDED.last_digest_sent_at
```

### `UpdateAccountSlackWebhook`
```sql
UPDATE accounts SET slack_webhook_url = $2
WHERE id = $1 AND tenant_id = current_setting('app.tenant_id', true)
```

### `ListTenantUsers`
```sql
SELECT id, tenant_id, kinde_sub, email, name, created_at, last_seen
FROM users
WHERE tenant_id = current_setting('app.tenant_id', true)
```
