# 2.16-4 — CloudFront + S3 Dashboard

> Serve the Expo static web build via S3 + CloudFront.
> The dashboard is a pure SPA — no server needed. Runtime config is baked in at build time via `EXPO_PUBLIC_*` args.

---

## How the dashboard works

- `npx expo export --platform web` → outputs static files to `dist/`
- `EXPO_PUBLIC_KINDE_ISSUER`, `EXPO_PUBLIC_KINDE_CLIENT_ID`, `EXPO_PUBLIC_API_URL` are baked in at build time
- `EXPO_PUBLIC_API_URL` must point to the production API App Runner URL
- All routes fall back to `index.html` (SPA routing) — CloudFront handles this via custom error response

## terraform/cloudfront.tf

```hcl
resource "aws_s3_bucket" "dashboard" {
  bucket = "axiaops-dashboard-prod"
}

resource "aws_s3_bucket_public_access_block" "dashboard" {
  bucket                  = aws_s3_bucket.dashboard.id
  block_public_acls       = true
  block_public_policy     = true
  ignore_public_acls      = true
  restrict_public_buckets = true
}

resource "aws_cloudfront_origin_access_control" "dashboard" {
  name                              = "axiaops-dashboard-oac"
  origin_access_control_origin_type = "s3"
  signing_behavior                  = "always"
  signing_protocol                  = "sigv4"
}

resource "aws_cloudfront_distribution" "dashboard" {
  enabled             = true
  default_root_object = "index.html"
  price_class         = "PriceClass_100"  # US + Europe only — cheapest

  origin {
    domain_name              = aws_s3_bucket.dashboard.bucket_regional_domain_name
    origin_id                = "s3-dashboard"
    origin_access_control_id = aws_cloudfront_origin_access_control.dashboard.id
  }

  default_cache_behavior {
    target_origin_id       = "s3-dashboard"
    viewer_protocol_policy = "redirect-to-https"
    allowed_methods        = ["GET", "HEAD"]
    cached_methods         = ["GET", "HEAD"]
    compress               = true

    forwarded_values {
      query_string = false
      cookies { forward = "none" }
    }

    # Short TTL for index.html so deploys propagate quickly
    min_ttl     = 0
    default_ttl = 60
    max_ttl     = 300
  }

  # SPA fallback — all 404s serve index.html with 200
  custom_error_response {
    error_code            = 404
    response_code         = 200
    response_page_path    = "/index.html"
    error_caching_min_ttl = 0
  }

  custom_error_response {
    error_code            = 403
    response_code         = 200
    response_page_path    = "/index.html"
    error_caching_min_ttl = 0
  }

  restrictions {
    geo_restriction { restriction_type = "none" }
  }

  viewer_certificate {
    cloudfront_default_certificate = true  # use *.cloudfront.net for now; add custom domain later
  }
}

# Allow CloudFront to read from S3
resource "aws_s3_bucket_policy" "dashboard" {
  bucket = aws_s3_bucket.dashboard.id
  policy = jsonencode({
    Version = "2012-10-17"
    Statement = [{
      Effect    = "Allow"
      Principal = { Service = "cloudfront.amazonaws.com" }
      Action    = "s3:GetObject"
      Resource  = "${aws_s3_bucket.dashboard.arn}/*"
      Condition = {
        StringEquals = {
          "AWS:SourceArn" = aws_cloudfront_distribution.dashboard.arn
        }
      }
    }]
  })
}
```

## Building and uploading the dashboard

The dashboard build requires the API URL baked in at build time:

```bash
# Build with production API URL
docker build \
  --build-arg EXPO_PUBLIC_KINDE_ISSUER=https://axiaops.kinde.com \
  --build-arg EXPO_PUBLIC_KINDE_CLIENT_ID=YOUR_CLIENT_ID \
  --build-arg EXPO_PUBLIC_API_URL=https://API_APPRUNNER_URL \
  --build-arg EXPO_PUBLIC_DEV_MODE=false \
  -f services/dashboard/Dockerfile \
  -t axiaops-dashboard-build \
  services/dashboard/

# Extract the built static files
docker create --name extract axiaops-dashboard-build
docker cp extract:/usr/share/nginx/html ./dist
docker rm extract

# Upload to S3
aws s3 sync ./dist s3://axiaops-dashboard-prod/ --delete

# Invalidate CloudFront cache
aws cloudfront create-invalidation \
  --distribution-id DISTRIBUTION_ID \
  --paths "/*"
```

## Update `.gitlab-ci.yml` deploy:production

The existing job pushes the dashboard image to ECR but doesn't upload to S3. Replace the dashboard App Runner update with an S3 sync:

```yaml
# Remove this (dashboard doesn't run on App Runner):
- aws apprunner update-service ... axiaops-dashboard ...

# Add this instead:
- docker build
    --build-arg EXPO_PUBLIC_KINDE_ISSUER=${EXPO_PUBLIC_KINDE_ISSUER}
    --build-arg EXPO_PUBLIC_KINDE_CLIENT_ID=${EXPO_PUBLIC_KINDE_CLIENT_ID}
    --build-arg EXPO_PUBLIC_API_URL=${EXPO_PUBLIC_API_URL}
    --build-arg EXPO_PUBLIC_DEV_MODE=false
    -f services/dashboard/Dockerfile
    -t dashboard-build
    services/dashboard/
- docker create --name extract dashboard-build
- docker cp extract:/usr/share/nginx/html ./dist
- docker rm extract
- aws s3 sync ./dist s3://${S3_DASHBOARD_BUCKET}/ --delete
- aws cloudfront create-invalidation
    --distribution-id ${CLOUDFRONT_DISTRIBUTION_ID}
    --paths "/*"
```
